## createPreferenceStore

createPreferenceStore validates loaded and assigned values before exposing a
stable snapshot.

{/* prettier-ignore */}
```typescript
const store = createPreferenceStore({
  defaultValue: defaultPreferences,
  persistence: {
    key: "chat:a11y-preferences",
    storage: window.localStorage,
    events: {
      subscribe(listener) {
        const handle = (event: StorageEvent) => listener(event);
        window.addEventListener("storage", handle);
        return () => window.removeEventListener("storage", handle);
      },
    },
  },
});

const unsubscribe = store.subscribe(() => {
  const preferences = store.getSnapshot();
  const configuration = preferencesToCoreConfiguration(preferences);
  applyConfiguration(configuration);
});

store.setPreferences({
  version: 1,
  preset: "balanced",
  streaming: "sentence",
  tools: "status",
});
```
